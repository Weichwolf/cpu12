cmake_minimum_required(VERSION 3.25)
project(cpu12_core VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/CompilerCache.cmake)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(language C CXX)
  if(NOT CMAKE_${language}_COMPILER_ID STREQUAL "Clang" OR
     NOT CMAKE_${language}_COMPILER_VERSION VERSION_GREATER_EQUAL 19 OR
     NOT CMAKE_${language}_COMPILER_VERSION VERSION_LESS 20)
    message(FATAL_ERROR "Use LLVM 19")
  endif()
endforeach()
include(cmake/Quality.cmake)
find_program(CPU12_CLANG_TIDY NAMES clang-tidy-19 REQUIRED)
find_program(CPU12_CLANG_FORMAT NAMES clang-format-19 REQUIRED)
set(CMAKE_CXX_CLANG_TIDY "${CPU12_CLANG_TIDY};--warnings-as-errors=*")
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
add_library(cpu12_core STATIC src/core.cpp src/decode.cpp src/indexed.cpp src/jit.cpp src/jit/emitter.cpp src/jit/indexed.cpp src/jit/stack.cpp src/runtime.cpp)
add_library(cpu12::core ALIAS cpu12_core)
target_include_directories(cpu12_core PUBLIC include PRIVATE src)
target_link_libraries(cpu12_core PUBLIC cpu12_quality)
target_compile_options(cpu12_core PUBLIC $<$<CONFIG:Release>:-march=native>)
target_link_options(cpu12_core PUBLIC -fuse-ld=lld)
file(GLOB_RECURSE CPU12_FORMAT_FILES CONFIGURE_DEPENDS include/*.hpp src/*.hpp src/*.cpp test/*.cpp test/*.hpp)
add_custom_target(format-check COMMAND "${CPU12_CLANG_FORMAT}" --dry-run --Werror ${CPU12_FORMAT_FILES} VERBATIM)
if(BUILD_TESTING)
  add_executable(cpu12-native-test test/native_test.cpp)
  target_link_libraries(cpu12-native-test PRIVATE cpu12::core)
  add_test(NAME native COMMAND cpu12-native-test)
  add_executable(cpu12-memory-test test/memory_test.cpp)
  target_link_libraries(cpu12-memory-test PRIVATE cpu12::core)
  add_test(NAME memory COMMAND cpu12-memory-test)
  add_executable(cpu12-word-test test/word_test.cpp)
  target_link_libraries(cpu12-word-test PRIVATE cpu12::core)
  add_test(NAME word COMMAND cpu12-word-test)
  add_executable(cpu12-stack-test test/stack_test.cpp test/stack_calls.cpp test/stack_bus.cpp)
  target_link_libraries(cpu12-stack-test PRIVATE cpu12::core)
  add_test(NAME stack COMMAND cpu12-stack-test)
  add_executable(cpu12-control-test test/control_test.cpp)
  target_link_libraries(cpu12-control-test PRIVATE cpu12::core)
  add_test(NAME control COMMAND cpu12-control-test)
  add_executable(cpu12-auto-indexed-test test/auto_indexed_test.cpp)
  target_link_libraries(cpu12-auto-indexed-test PRIVATE cpu12::core)
  add_test(NAME auto-indexed COMMAND cpu12-auto-indexed-test)
  add_executable(cpu12-indexed-test test/indexed_test.cpp)
  target_link_libraries(cpu12-indexed-test PRIVATE cpu12::core)
  add_test(NAME indexed COMMAND cpu12-indexed-test)
  add_executable(cpu12-memory-api-test test/memory_api_test.cpp)
  target_link_libraries(cpu12-memory-api-test PRIVATE cpu12::core)
  add_test(NAME memory-api COMMAND cpu12-memory-api-test)
endif()
